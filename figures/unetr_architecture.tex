\foreach \i [evaluate=\i as \y using {2 + 2.8 - \i*\tspacing}] in {1,...,12} {

}

% Extraction indicators
\foreach \i/\lab in {3/$z_3$, 6/$z_6$, 9/$z_9$, 12/$z_{12}$} {
