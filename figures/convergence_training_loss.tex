% convergence_training_loss.tex
%
% Training loss convergence: nnU-Net (top) vs. UNETR (bottom).
% Compile:  pdflatex convergence_training_loss.tex  (twice for pgfplots aux)
%
% Requires: pgfplots >= 1.16, pgfplotstable, pgfplots fillbetween library
% Data:  data/nnunet_loss.csv   columns: epoch,mean,lower,upper
%        data/unetr_loss.csv    columns: epoch,mean,lower,upper

\documentclass[border=8pt]{standalone}

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{fillbetween}
\usepackage{pgfplotstable}
\usetikzlibrary{calc}

% ---------------------------------------------------------------------------
% Colours
% ---------------------------------------------------------------------------
\definecolor{cNNunet}{HTML}{E07B39}   % orange
\definecolor{cUNETR} {HTML}{3A86C8}   % blue
\definecolor{cMarker}{HTML}{CC3333}   % red

% ---------------------------------------------------------------------------
% Styles
% ---------------------------------------------------------------------------
\pgfplotsset{
  lossaxis/.style = {
    width  = 11cm,
    height = 5.5cm,
    tick label style = {font = \scriptsize},
    label style      = {font = \footnotesize},
    xlabel = {Epoch},
    ylabel = {Dice + CE Loss},
    grid      = major,
    grid style= {gray!25, line width=0.4pt},
    legend style = {
      font          = \scriptsize,
      at            = {(0.97, 0.97)},
      anchor        = north east,
      draw          = black!30,
      fill          = white,
      fill opacity  = 0.88,
      text opacity  = 1,
      legend cell align = left,
      row sep       = 1pt,
    },
    tick align  = outside,
    tick pos    = left,
    line width  = 0.6pt,
    axis line style = {line width=0.8pt},
    minor tick num  = 0,
  },
  % Secondary x-axis drawn on top of a main axis.
  % Place with: at={(mainaxis.north west)}, topaxis
  topaxis/.style = {
    width  = 11cm,
    height = 5.5cm,
    anchor = north west,
    axis x line* = top,
    axis y line  = none,
    ytick        = \empty,
    tick label style = {font = \scriptsize},
    xlabel style     = {font = \scriptsize},
    tick align  = outside,
    line width  = 0.6pt,
    axis line style = {line width=0.8pt},
    minor tick num  = 0,
    scaled x ticks  = false,
  },
}

% ---------------------------------------------------------------------------
% Load data
% ---------------------------------------------------------------------------
\pgfplotstableread[col sep=comma]{data/nnunet_loss.csv}\nnunetloss
\pgfplotstableread[col sep=comma]{data/unetr_loss.csv}\unetrloss

% ---------------------------------------------------------------------------
\begin{document}
\begin{tikzpicture}

% ===========================================================================
% (a)  nnU-Net — main axis
% ===========================================================================
\begin{axis}[
  name   = nnunet,
  at     = {(0,0)},
  anchor = north west,
  lossaxis,
  xmin = -10,    xmax = 1009,
  ymin = 0.10,   ymax = 0.82,
  ytick = {0.2, 0.4, 0.6, 0.8},
]
  % ±1 std band
  \addplot[name path=nn_up, draw=none, forget plot]
    table[x=epoch, y expr={\thisrow{upper}+1}]{\nnunetloss};
  \addplot[name path=nn_lo, draw=none, forget plot]
    table[x=epoch, y expr={\thisrow{lower}+1}]{\nnunetloss};
  \addplot[fill=cNNunet, fill opacity=0.18, draw=none, forget plot]
    fill between[of=nn_up and nn_lo];

  % mean curve
  \addplot[color=cNNunet, line width=1.6pt]
    table[x=epoch, y expr={\thisrow{mean}+1}]{\nnunetloss};
  \addlegendentry{Mean (5 folds)}

  % std legend entry
  \addlegendimage{fill=cNNunet, fill opacity=0.4, draw=none, area legend}
  \addlegendentry{$\pm 1$ Std}

  % epoch-100 line
  \addplot[color=cMarker, dashed, line width=1.2pt, forget plot]
    coordinates {(100, 0.10) (100, 0.82)};
  \addlegendimage{color=cMarker, dashed, line width=1.2pt}
  \addlegendentry{Epoch 100}

  % annotation
  \draw[cMarker, dotted, line width=0.7pt]
    (axis cs:0, 0.2316) -- (axis cs:100, 0.2316);
  \node[anchor=west, text=cMarker, font=\scriptsize]
    (nn_label) at (axis cs:120, 0.285) {Loss\,$=0.232$ (ep\,100)};
  \draw[->, cMarker, line width=0.7pt]
    (nn_label.west) -- (axis cs:102, 0.242);
\end{axis}

% (a) iteration axis — overlaps nnunet exactly
\begin{axis}[
  name   = nn_topax,
  at     = {(nnunet.north west)},
  topaxis,
  xmin   = -2500,
  xmax   = 252250,
  xlabel = {Iterations\enspace ($1\,\mathrm{ep} = 250\,\mathrm{iters}$)},
  xtick  = {0, 50000, 100000, 150000, 200000, 250000},
  xticklabels = {$0$, $50\mathrm{k}$, $100\mathrm{k}$,
                 $150\mathrm{k}$, $200\mathrm{k}$, $250\mathrm{k}$},
  ymin = 0.10, ymax = 0.82,
]
\end{axis}

% (a) panel title — above nn_topax.outer north (above ticks + xlabel)
\node[anchor=south, yshift=4pt, font=\footnotesize]
  at (nn_topax.outer north)
  {(a)\enspace nnU-Net\enspace
   (250\,iters/epoch,\; $\mathrm{bs}{=}2$,\; 1000\,epochs)};

% ===========================================================================
% (b)  UNETR — 3 cm below nnunet.outer south (outer includes tick labels/xlabel)
% ===========================================================================
\begin{axis}[
  name   = unetr,
  at     = {($(nnunet.south west)+(0,-3.5cm)$)},
  anchor = north west,
  lossaxis,
  xmin = -1,   xmax = 161,
  ymin = 0.18, ymax = 2.30,
  ytick = {0.5, 1.0, 1.5, 2.0},
]
  % ±1 std band
  \addplot[name path=ut_up, draw=none, forget plot]
    table[x=epoch, y=upper]{\unetrloss};
  \addplot[name path=ut_lo, draw=none, forget plot]
    table[x=epoch, y=lower]{\unetrloss};
  \addplot[fill=cUNETR, fill opacity=0.18, draw=none, forget plot]
    fill between[of=ut_up and ut_lo];

  % mean curve
  \addplot[color=cUNETR, line width=1.6pt]
    table[x=epoch, y=mean]{\unetrloss};
  \addlegendentry{Mean (5 folds)}

  % std legend entry
  \addlegendimage{fill=cUNETR, fill opacity=0.4, draw=none, area legend}
  \addlegendentry{$\pm 1$ Std}

  % epoch-100 line
  \addplot[color=cMarker, dashed, line width=1.2pt, forget plot]
    coordinates {(100, 0.18) (100, 2.30)};
  \addlegendimage{color=cMarker, dashed, line width=1.2pt}
  \addlegendentry{Epoch 100}

  % annotation
  \draw[cMarker, dotted, line width=0.7pt]
    (axis cs:0, 0.2755) -- (axis cs:100, 0.2755);
  \node[anchor=east, text=cMarker, font=\scriptsize]
    (ut_label) at (axis cs:92, 0.44) {Loss\,$=0.276$ (ep\,100)};
  \draw[->, cMarker, line width=0.7pt]
    (ut_label.east) -- (axis cs:98, 0.282);
\end{axis}

% (b) iteration axis — overlaps unetr exactly
\begin{axis}[
  name   = ut_topax,
  at     = {(unetr.north west)},
  topaxis,
  xmin   = -501,
  xmax   = 80661,
  xlabel = {Iterations\enspace ($1\,\mathrm{ep} \approx 501\,\mathrm{iters}$)},
  xtick  = {0, 20000, 40000, 60000, 80000},
  xticklabels = {$0$, $20\mathrm{k}$, $40\mathrm{k}$,
                 $60\mathrm{k}$, $80\mathrm{k}$},
  ymin = 0.18, ymax = 2.30,
]
\end{axis}

% (b) panel title — above ut_topax.outer north
\node[anchor=south, yshift=4pt, font=\small]
  at (ut_topax.outer north)
  {(b)\enspace UNETR\enspace
   (501\,iters/epoch,\; $\mathrm{bs}{=}2$,\;
    grad-accum\,$=4$,\; early stop)};

% Force standalone bounding box to include full UNETR panel bottom
\path[draw=none] (unetr.outer south west) rectangle (unetr.outer north east);

\end{tikzpicture}
\end{document}
